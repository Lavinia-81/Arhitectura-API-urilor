Ex5. Detectare IP-uri multiple — anti-scraping
Cerință: Simulează 3 IP-uri diferite pentru aceeași cheie API.

-----

R => Logica anti-scraping
// src/rateLimiter/ipDetection.ts
import { redis } from "../redis";
export async function detectMultipleIPs(apiKey: string, ip: string) {
  const key = `ips:${apiKey}`;
  await redis.sadd(key, ip);
  const count = await redis.scard(key);

  if (count > 3) {
    console.warn(`ALERT: API key ${apiKey} used from ${count} IPs`);
    return true;
  }
  return false;
}

-----


R => Integrare în endpoint
const ip = req.ip;
const suspicious = await detectMultipleIPs(apiKey, ip);

if (suspicious) {
  reply.header("X-Scraping-Alert", "true");
}
