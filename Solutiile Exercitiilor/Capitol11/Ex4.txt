// src/billing/usage.ts
import { prisma } from "../db";
export async function getMonthlyUsage(userId: string) {
  const start = new Date();
  start.setDate(1);
  return prisma.usage.count({
    where: {
      user_id: userId,
      timestamp: { gte: start }
    }
  });
}
