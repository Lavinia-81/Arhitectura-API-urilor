Ex6. Adaugă full‑text search în limba română
-----

R => 1. Activezi dicționarul românesc
CREATE EXTENSION IF NOT EXISTS unaccent;

-----

R => 2. Adaugi coloană tsvector
ALTER TABLE work ADD COLUMN search tsvector;

-----

R => 3. Populezi câmpul
UPDATE work
SET search = to_tsvector('romanian', unaccent(title || ' ' || author));

-----

R => 4. Trigger automat
CREATE FUNCTION work_search_update() RETURNS trigger AS $$
BEGIN
  NEW.search := to_tsvector('romanian', unaccent(NEW.title || ' ' || NEW.author));
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER work_search_trigger
BEFORE INSERT OR UPDATE ON work
FOR EACH ROW EXECUTE FUNCTION work_search_update();

-----

R => 5. Query full‑text
SELECT *
FROM work
WHERE search @@ to_tsquery('romanian', 'eminescu');
Rezultatul: toate lucrările legate de Eminescu.
