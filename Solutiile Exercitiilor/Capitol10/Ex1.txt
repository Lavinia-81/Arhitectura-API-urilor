Ex1. Pino + middleware de logging

-----


R => Configurare Pino în Fastify

// src/index.ts
import Fastify from "fastify";
const app = Fastify({
  logger: true
});
-----

R => Middleware care loghează request_id, metodă, path, status, durată

// src/middlewares/request-logging.ts
export async function requestLogging(app) {
  app.addHook("onRequest", async (req) => {
    // dacă nu ai deja request_id, îl poți genera
    req.request_id = req.request_id || `req_${Date.now()}`;
    (req.log as any).info({
      request_id: req.request_id,
      method: req.method,
      path: req.url
    }, "Incoming request");
  });

  app.addHook("onResponse", async (req, reply) => {
    const duration = reply.getResponseTime();
    (req.log as any).info({
      request_id: req.request_id,
      method: req.method,
      path: req.url,
      status: reply.statusCode,
      duration
    }, "Request completed");
  });
}

-----

Și în index.ts:
import { requestLogging } from "./middlewares/request-logging";
await requestLogging(app);
